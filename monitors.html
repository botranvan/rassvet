panels/default.yaml
# instances: the libvirt exec script
- dashboard: cloud.*
  title: Instances
  kind: instance        # or physical
  panels:
    - name: cpu
      title: CPU usage
      unit: percent
      queries:
        - query: >-
            100 * sum(rate(libvirt_cpu_user{
              guest_id="$guest_id"}[$__rate_interval]))
            / 1e9 / max(libvirt_cpu_number{...})
          legend: user
